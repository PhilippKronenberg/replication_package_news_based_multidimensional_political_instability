\begin{section}[Introduction]{Introduction}\label{motivation_PI}


\begin{frame}[label=overview]{Motivation}

\textbf{Political Stability}
    \begin{itemize}
        \item  Political instability remains a central obstacle to economic development in West and Central Africa.
        \item Political instability erodes investor confidence, weakens state institutions, and increases the risk of violent conflict, thereby constraining growth and poverty reduction.
        \item Accurate and timely assessment of political stability is essential for early-warning systems and evidence-based policy design in the region.
    \end{itemize}
\vspace{0.25cm}
\textbf{Challenges}
\begin{itemize}
    \item \textbf{Data Scarcity:} Traditional data sources are often limited%; our index fills the gaps.
    \item \textbf{Low Frequency:} Conventional data is primarily annual or quarterly %; our index offers high-frequency updates for better decision-making.
    \item \textbf{Timeliness:} Event- or survey-based indicators are retrospective rather than forward-looking.
    \item \textbf{Dimensionality:} Different sources of political instability might have different implications
\end{itemize}

\end{frame}

\begin{frame}[label=overview]{Research Question}

\textbf{Goal}\\
Can we extract information from newspapers (text data) to obtain a high-frequency signal of the political stability in West and Central African countries? \\
\vspace{0.25cm}
\textbf{In Detail}
    \begin{itemize}
        \item Can we measure different dimensions of political stability with newspaper data?
        \item Do our indicators correctly depict past political instability events?
        \item How do different political instability shocks transmit on the macroeconomy?
    \end{itemize}
    \vspace{0.25cm}
    \textbf{Benefits}
        \begin{itemize}
        \item We fill the gap of missing statistical data with high-frequency indicator
        \item Such a measure is new for the Western and Central African region
        \item We look at more diverse dimensions of political instability than existing indicators
        \item Our measure is forward looking and can be used for policy decisions
    \end{itemize}
\end{frame}

\begin{frame}[label=overview]{The Study Setup}
    \begin{itemize}
        \item \textbf{Data:} Use leading global news sources in English from Factiva Dow Jones news aggregator \vspace{0.2cm}
        \item \textbf{Method:} Use keyword search to create a count measure representing a sentiment \vspace{0.2cm}
        \item \textbf{Dimensions:} Politically motivated violence, Mass civil protest, instability within the political regime, instability of the political regime (following \cite{JONGAPIN2009}) \vspace{0.2cm}
        \item \textbf{Coverage:} 17 West and Central African countries: CEMAC (Cameroon, Central African Republic, Chad, Equatorial Guinea, Gabon, Republic of Congo), WAEMU (Benin, Burkina Faso, Cote d’Ivoire, Guinea/Bissau, Mali, Niger, Senegal, Togo), West-Sahel (Mali, Niger, Burkina Faso, Chad, Senegal, Mauritania, Nigeria)
    \end{itemize}
\end{frame}

\section{Methodology}\label{methodology_PI}


\begin{frame}{Keyword search}\label{keywords_search} 
\small
Transparent keyword-based search, tailored to four dimensions \citep{JONGAPIN2009}. Inclusive criteria (strings/tags) to reduce false negatives; exclusive criteria to reduce false positives.
%\textbf{Keyword search}
\begin{itemize}
    \item \textbf{Location tag:} Country name
    \item \textbf{Subject included tag:}
        \begin{itemize}
        {\scriptsize 
            \item Domestic Politics, National/Public Security, Civil Unrest, Armed Forces, Terrorism
            \item Domestic Politics, National/Public Security, Elections, impeachment, Political Appointments/Terminations 
            }
        \end{itemize}
        \item \textbf{Subject excluded tag:} \\
        {\scriptsize Exclude any non-articles or lifestyle and routine news (weather, sports, images, etc.)
        }
        \item \textbf{Keywords excluded:} 
        \begin{itemize}
        {\scriptsize 
            \item \cite{CALDARA2022}: movie*, film*, museum*, anniversar*, obituar*, memorial*, arts, book, books, memoir*, “price war”, game, story, history, veteran*, tribute*, sport, music, racing, cancer, “real estate”, mafia, trial, tax
            \item \cite{BARRETT2022}: “vote of protest”, “protest vote”, protestant*, do not use “demonstration”
            }
        \end{itemize}
        \item \textbf{Keywords included:} Include keywords related to the topic
        \item \textbf{Search operators:} and, or, not, near, atleast, within, truncation sign, first x words, word count
    \end{itemize}

    \hyperlink{search_details}{\beamerreturnbutton{Search Details}}
\end{frame}

\begin{frame}{Political Instability Indices (\cite{BARRETT2022})}

\begin{columns}[T,onlytextwidth]

% ===================== Index A =====================
\begin{column}{0.5\textwidth}
\small
\textbf{Index A: Regional Normalization}

\vspace{0.3em}
\begin{equation*}
\text{IndexA}_{it}
= \frac{x_{it}}{\frac{1}{12}\sum_{j=1}^{12} z_{t-j}}
\times \frac{100}{\bar{x}_i/\bar{z}}
\end{equation*}

%\vspace{-0.6em}
\begin{itemize}\setlength\itemsep{0.4em}
\item $x_{it}$: instability-related articles in country $i$ at time $t$
\item $z_t$: total contemporary articles across all countries
\item Normalized by regional coverage and scaled to mean 100
\item Corrects for common regional media trends
\item Robust over time, but not comparable in importance across countries
\end{itemize}
\end{column}

% ===================== Index B =====================
\begin{column}{0.5\textwidth}
\small
\textbf{Index B: Country-Specific Share}

\vspace{0.3em}
\begin{equation*}
\text{IndexB}_{it}
= \frac{x_{it}}{\frac{1}{12}\sum_{j=1}^{12} y_{i,t-j}}
\end{equation*}

%\vspace{-0.6em}
\begin{itemize}\setlength\itemsep{0.4em}
\item $y_{it}$: contemporary articles in country $i$ at time $t$
\item Normalized by a country-specific rolling mean
\item Smoother series with time-varying sensitivity
\item Up-weights small events during low coverage periods
\item Potential issues: inflated signals under low coverage and endogeneity of news coverage to political instability
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{section}[Conclusion]{Conclusion}

\begin{frame}{Conclusion}\label{conclusion_PI}

\small
\begin{itemize}\setlength\itemsep{0.6em}

\item The four dimensions reflect distinct yet interrelated facets of instability. They differ in persistence, cluster around different political events, and exhibit both cross-border and cross-dimension contagion. Political violence emerges as a persistent background risk, while protest- and regime-related instability is more episodic.

\item A VAR analysis shows heterogeneous economic transmission: political violence acts as localized supply and risk shocks; protests as short-lived uncertainty; within-regime instability as persistent uncertainty; and regime instability as a broad confidence shock.

\item Limitations are that the indicators reflect international media coverage rather than a full census of events and that they are sensitive to keyword choices.

\item Despite limitations, the indicators offer a flexible and policy-relevant tool for monitoring political instability in data-scarce environments.

\end{itemize}

\end{frame}

\end{section}
